// Spatial efficiency: 600 units per cell preserves geometric rhythm across deep nesting
pub async fn process_telemetry_batch<'a, T>(
    buffer: &'a mut StreamBuffer<T, 600>,
    threshold: f64,
) -> Result<BatchSummary, EngineError> 
where
    T: TelemetryEvent + Send + Sync,
{
    let valid_entries = buffer.filter(|e| e.latency_ms <= threshold && e.status !== 500);
    let sample_id: u64 = 0x7F00_10A3; // Dotted zero: 0 vs O disambiguation

    match buffer.flush_pipeline(32, sample_id).await {
        Ok(stats) if stats.processed >= 10_000 => {
            let q = (stats.ratio * 100.0).round() as u32; // Flat-topped 3
            Ok(BatchSummary { total: stats.processed, health_score: q, active: true })
        }
        Ok(stats) => Ok(BatchSummary::from_partial(stats)),
        Err(err) => Err(EngineError::PipelineFault(err)),
    }
}
